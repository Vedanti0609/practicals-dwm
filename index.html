Python DAG program

from airflow import DAG
from airflow.operators.python import PythonOperator
from datetime import datetime

def extract_data():
    print("Extracting sales data...")

def validate_data():
    print("Validating sales data...")

def transform_data():
    print("Transforming sales data...")

def generate_report():
    import os
    output_dir = os.path.expanduser("~/airflow/output")
    os.makedirs(output_dir, exist_ok=True)
    output_file = os.path.join(output_dir, "sales_summary.txt")

    with open(output_file, "w") as f:
        f.write("Sales Summary Report\n")
        f.write("====================\n")
        f.write("Total Sales: 1000\n")
        f.write("Total Orders: 100\n")

    print(f"Report generated: {output_file}")

def finish_pipeline():
    print("Sales pipeline completed successfully!")

with DAG(
    dag_id="student_sales_pipeline",
    start_date=datetime(2026, 9, 9),
    schedule="*/5 * * * *",
    catchup=False,
) as dag:

    extract = PythonOperator(
        task_id="extract_data",
        python_callable=extract_data,
    )

    validate = PythonOperator(
        task_id="validate_data",
        python_callable=validate_data,
    )

    transform = PythonOperator(
        task_id="transform_data",
        python_callable=transform_data,
    )

    report = PythonOperator(
        task_id="generate_report",
        python_callable=generate_report,
    )

    finish = PythonOperator(
        task_id="finish_pipeline",
        python_callable=finish_pipeline,
    )

    extract >> validate >> transform >> report >> finish

from pyspark.sql import SparkSession

spark = SparkSession.builder \
    .appName("Parallel Processing Practical") \
    .master("local[*]") \
    .getOrCreate()

sc = spark.sparkContext

numbers = list(range(1, 21))
rdd = sc.parallelize(numbers, 4)

print("Number of partitions:", rdd.getNumPartitions())

squared_rdd = rdd.map(lambda x: x * x)

print("Squared values:", squared_rdd.collect())

total = squared_rdd.reduce(lambda x, y: x + y)
print("Sum of squared values:", total)

spark.stop()

# Save the Python program, then run:

from pyspark.sql import SparkSession
from pyspark.sql.functions import col

spark = SparkSession.builder \
    .appName("Query Optimization Practical") \
    .master("local[*]") \
    .getOrCreate()

data = [
    (1, "Laptop", "West", 55000),
    (2, "Mouse", "East", 800),
    (3, "Keyboard", "West", 1500),
    (4, "Monitor", "South", 12000),
    (5, "Laptop", "North", 60000),
    (6, "Mouse", "West", 900),
    (7, "Monitor", "East", 11000),
    (8, "Keyboard", "North", 1700)
]

columns = ["order_id", "product", "region", "sales"]
df = spark.createDataFrame(data, columns)

df.show()

df.write.mode("overwrite") \
    .partitionBy("region") \
    .parquet("sales_partitioned")

print("Partitioned data saved.")

sales_df = spark.read.parquet("sales_partitioned")
west_df = sales_df.filter(col("region") == "West")

west_df.show()

print("Execution Plan:")
west_df.explain(True)

west_df.cache()
print("West region count:", west_df.count())

spark.catalog.clearCache()
spark.stop()

5(c) Fault tolerance

nano fault_tolerance.py

from pyspark.sql import SparkSession

spark = SparkSession.builder \
    .appName("Fault Tolerance Practical") \
    .master("local[4]") \
    .config("spark.task.maxFailures", "4") \
    .getOrCreate()

sc = spark.sparkContext
rdd = sc.parallelize(range(1, 101), 4)

def process_partition(iterator):
    values = list(iterator)

    if 1 in values:
        raise Exception("Simulated task failure")

    return iter([x * 2 for x in values])

try:
    result = rdd.mapPartitions(process_partition).collect()
    print("Result:", result)

except Exception as error:
    print("The task failed as expected.")
    print("Error:", error)

spark.stop()

# Run:
spark-submit fault_tolerance.py

    .appName("Checkpoint Practical") \
